\section{2. Baseline NeuMF}

\begin{frame}{NeuMF --- Neural Matrix Factorization (He et al., 2017)}
  \begin{columns}[T]
    \column{0.54\textwidth}
      \textbf{Ý tưởng:} Kết hợp 2 nhánh:
      \begin{itemize}
        \item \textbf{GMF} (Generalized Matrix Factorization):
          tích Hadamard $\mathbf{p}_u \odot \mathbf{q}_i$
          $\to$ nắm bắt tương tác tuyến tính
        \item \textbf{MLP} (Multi-Layer Perceptron):
          concat $[\mathbf{p}_u;\mathbf{q}_i]$ qua các lớp FC
          $\to$ nắm bắt quan hệ phi tuyến
      \end{itemize}
      
      \textbf{Dự đoán:} $\hat{y}_{ui} = \sigma\!\bigl(\mathbf{h}^\top
          [\phi_{\text{GMF}};\phi_{\text{MLP}}]\bigr)$

\begin{warnbox}[title={\faExclamationTriangle\ Giới hạn với tin tức}]
        \footnotesize
        NeuMF học \textbf{embedding theo ID} của user và item.
        Tin tức \textit{mới} chưa có ID $\to$ mô hình \textbf{không dự đoán được}
        (cold-start problem).
      \end{warnbox}

\section{3. Mô hình NRMS}

\begin{frame}{NRMS --- Neural News Recommendation with Multi-Head Self-Attention (Wu et al., EMNLP 2019)}
  \begin{columns}[T]
    \column{0.56\textwidth}
      \textbf{Hai encoder độc lập:}
      \begin{enumerate}
        \item \textbf{News Encoder}: mã hóa từng bài báo thành vector $\mathbf{n} \in \mathbb{R}^d$
        \item \textbf{User Encoder}: tổng hợp lịch sử thành $\mathbf{u} \in \mathbb{R}^d$

      \vspace{0.3em}

        $h$ = số attention heads\\
        Trong code: $d{=}256$, $h{=}16$, $T{=}30$, $L{=}50$
      \end{keybox}

\begin{frame}{Multi-Head Self-Attention (MH-SA) --- chi tiết}
  \begin{columns}[T]
    \column{0.50\textwidth}
      \textbf{Công thức Scaled Dot-Product Attention:}
      \[
        \text{Attn}(Q,K,V) = \text{softmax}\!\!\left(\frac{QK^\top}{\sqrt{d_k}}\right)\!V
      \]

      \textbf{Ký hiệu:}
      \begin{itemize}
        \item $Q \in \mathbb{R}^{L \times d_k}$: Query (truy vấn)
        \item $K \in \mathbb{R}^{L \times d_k}$: Key (khóa tra cứu)
        \item $V \in \mathbb{R}^{L \times d_v}$: Value (giá trị)
        \item $d_k = d/h$: chiều mỗi head, dùng để \textit{scale}
          (chia $\sqrt{d_k}$ tránh gradient vanish khi $d_k$ lớn)
      \end{itemize}

      \textbf{Multi-Head:} $
        \text{MH}(Q,K,V) = \text{Concat}(\text{head}_1,\dots,\text{head}_h)W^O
$

    \column{0.47\textwidth}
      \begin{warnbox}[title={\faExclamationTriangle\ Bottleneck (điểm nghẽn)}]
        \footnotesize
        Ma trận $QK^\top \in \mathbb{R}^{L \times L}$
        phải được lưu toàn bộ trong VRAM.\\[0.3em]
        $L=50$: $50 \times 50 \times h = 40\,000$ floats --- OK\\
        $L=200$: $200 \times 200 \times h \to$ \textbf{tăng $16\times$}\\
        $L=500$: $500^2 \times 16 \approx 4$ M floats/batch
        $\to$ $\mathcal{O}(L^2)$ memory growth:
training dễ OOM khi $L$ hoặc batch size lớn.
      \end{warnbox}
      \vspace{0.2em}
      \begin{exampleblock}{Additive Attention Pooling}
        \footnotesize
        Sau MH-SA, tổng hợp $L$ vector thành 1:\\
        $\alpha_i = \text{softmax}(\mathbf{v}^\top \tanh(W\mathbf{h}_i))$\\
        $\mathbf{r} = \sum_i \alpha_i \mathbf{h}_i$\\[0.2em]
        Dùng cả trong News Encoder lẫn User Encoder.
      \end{exampleblock}
  \end{columns}
\end{frame}

\section{4. Mô hình NRAGLS}

\begin{frame}{Gated Linear Attention (GLA)}
  \begin{columns}[T]
    \column{0.54\textwidth}
      \textbf{Ý tưởng cốt lõi:} Xấp xỉ softmax bằng \textbf{kernel map} $\phi(\cdot)$:
      \[
        \text{LinearAttn}(Q,K,V) = \phi(Q)\bigl(\phi(K)^\top V\bigr)
      \]

      \textbf{Tính tuyến tính:}
      \begin{itemize}
        \item Thứ tự nhân \textit{quan trọng}:\\
          $\phi(K)^\top V \in \mathbb{R}^{d \times d}$ --- tính \textit{trước}!
        \item Không còn ma trận $L \times L$
        \item Phức tạp: $\mathcal{O}(L \cdot d^2)$ thay vì $\mathcal{O}(L^2 \cdot d)$
        \item Khi $L \gg d$: \hl{tiết kiệm cực lớn}
      \end{itemize}

      \vspace{0.3em}
      \textbf{Kernel map:} $\phi(x) = \text{ELU}(x) + 1$\\
      {\footnotesize ELU = Exponential Linear Unit, đảm bảo $\phi(x) > 0$
       (cần để xấp xỉ softmax hợp lệ)}

    \column{0.43\textwidth}
      \begin{center}
      \begin{tikzpicture}[scale=0.75]
        % Standard Attention
        \node[font=\footnotesize\bfseries, nrmsblue] at (1.5,5.2) {Standard Attention};
        \draw[fill=nrmsblue!15, draw=nrmsblue, rounded corners=3pt]
          (0,4.3) rectangle (3,4.9) node[midway,font=\scriptsize]{$Q\in\mathbb{R}^{L\times d}$};
        \draw[fill=nrmsblue!15, draw=nrmsblue, rounded corners=3pt]
          (0,3.5) rectangle (3,4.1) node[midway,font=\scriptsize]{$K\in\mathbb{R}^{L\times d}$};
        \draw[fill=warnred!30, draw=warnred, rounded corners=3pt]
          (0,2.7) rectangle (3,3.3) node[midway,font=\scriptsize]
          {$QK^\top\!\in\!\mathbb{R}^{L\times L}$};
        \node[font=\tiny, warnred] at (1.5,2.4) {$\uparrow$ bottleneck!};

        % Linear Attention
        \node[font=\footnotesize\bfseries, nraglsorange] at (6.5,5.2) {Linear Attention};
        \draw[fill=nraglsorange!15, draw=nraglsorange, rounded corners=3pt]
          (4.8,4.3) rectangle (8.2,4.9) node[midway,font=\scriptsize]{$\phi(K)^\top V \in \mathbb{R}^{d\times d}$};
        \draw[fill=accentteal!20, draw=accentteal, rounded corners=3pt]
          (4.8,3.5) rectangle (8.2,4.1) node[midway,font=\scriptsize]{$\phi(Q)\cdot(\phi(K)^\top V)$};
        \node[font=\tiny, accentteal] at (6.5,3.2) {$\uparrow$ $\mathcal{O}(Ld^2)$ --- OK!};
      \end{tikzpicture}
      \end{center}
  \end{columns}
\end{frame}

% ─────────────────────────────────────────────────────────────────────────────
\begin{frame}{Gating Mechanism --- Lọc nhiễu lịch sử}
  \begin{columns}[T]
    \column{0.50\textwidth}
    \small
      \begin{warnbox}[title={\faExclamationTriangle\ Vấn đề}]
        Lịch sử click chứa \textbf{noise} (nhiễu):\\
        click nhầm, clickbait, bài đọc theo thói quen
        không phản ánh sở thích thật sự.
      \end{warnbox}
      \vspace{0.2em}
      \textbf{Giải pháp --- Sigmoid Gate:}
        $
        g = \sigma(W_g x + b_g), \quad g \in (0,1)^d 
        $
        $
        K_{\text{gated}} = K \odot g, \quad
        V_{\text{gated}} = V \odot g
        $

      \textbf{Ý nghĩa:}
      \begin{itemize}
        \item $g_i \approx 0$: \textcolor{warnred}{chặn} chiều nhiễu
        \item $g_i \approx 1$: \textcolor{accentteal}{giữ} chiều quan trọng
        \item $\odot$: phép nhân từng phần tử (element-wise)
        \item Tương tự \textit{Forget Gate} trong LSTM
      \end{itemize}

    \column{0.47\textwidth}
      \begin{exampleblock}{SGLU --- Simplified Gated Linear Unit}
        \footnotesize
        Thay FFN (Feed-Forward Network) truyền thống bằng:\\[0.3em]
        $\text{SGLU}(x) = \text{SiLU}(W_1 x) \odot (W_2 x)$\\[0.3em]
        \textit{SiLU} (Sigmoid Linear Unit): $x \cdot \sigma(x)$\\
        Ưu điểm: ít tham số hơn, thông tin chảy qua
        2 nhánh song song, kết hợp tuyến tính \& phi tuyến.
      \end{exampleblock}
      \vspace{0.3em}
      \begin{exampleblock}{RMSNorm --- Root Mean Square Norm}
        \footnotesize
        $\text{RMSNorm}(x) = \dfrac{x}{\text{RMS}(x)}\cdot\gamma$
        \quad với $\text{RMS}(x)=\sqrt{\tfrac{1}{d}\sum_i x_i^2}$\\[0.3em]
        Nhẹ hơn LayerNorm: bỏ bước tính mean.
        Tốc độ nhanh hơn ~$15\%$.
      \end{exampleblock}
  \end{columns}
\end{frame}

% ─────────────────────────────────────────────────────────────────────────────
\begin{frame}{Bổ sung: Recency-Decayed Gating (Suy giảm theo thời gian)}
  \begin{columns}[T]
    \column{0.54\textwidth}
      \begin{warnbox}[title={Hạn chế của NRAGLS gốc}]
      \small
        Gate $g = \sigma(Wx+b)$ chỉ dựa trên \textit{nội dung},
        hoàn toàn bỏ qua \textbf{thứ tự thời gian}.
        Bài đọc từ 6 tháng trước bằng bài đọc hôm qua!
      \end{warnbox}

      \vspace{0.1em}
      \textbf{Dùng Exponential Decay:}
      $
        g_{\text{final}}(i) = g(i) \cdot \alpha^{(L-1-i)}
      $
      \begin{itemize}
        \item $i$: vị trí trong lịch sử (0 = cũ nhất, $L{-}1$ = mới nhất)
        \item $\alpha \in (0,1)$: \textbf{learnable parameter}, khởi tạo $\approx 0.95$
        \item $\alpha^0=1$: bài mới nhất không bị suy giảm
        \item $\alpha^{L-1}\to 0$: bài cũ bị triệt tiêu dần
      \end{itemize}

    \column{0.43\textwidth}
      \begin{keybox}[title={Phân tích Half-life}]
        \footnotesize
        Half-life (thời gian bán rã) theo vị trí:
        $
          t_{1/2} = \frac{-1}{\ln\alpha} \text{ bước}
        $\\
        Nếu $\alpha=0.95$: $t_{1/2} \approx 13.5$ bước\\[0.3em]
        $\Rightarrow$ Bài đọc cách đây 13 vị trí
        chỉ còn $50\%$ trọng số.
      \end{keybox}
      \vspace{0.2em}
      \begin{goodbox}[title={\faCheckCircle\ Ưu điểm}]
        \footnotesize
        \begin{itemize}
          \item Không tăng complexity: vẫn $\mathcal{O}(Ld^2)$
          \item $\alpha$ tự học trong training
          \item Phản ánh hành vi thực: tin gần đây quan trọng hơn
        \end{itemize}
      \end{goodbox}
  \end{columns}
\end{frame}

\section{5. Kỹ thuật huấn luyện}

\begin{frame}{Hàm mất mát: InfoNCE --- Tại sao không dùng BCE hay BPR?}
  \begin{columns}[T]
    \column{0.50\textwidth}
      \textbf{InfoNCE} (Information Noise Contrastive Estimation):
      $
        \mathcal{L} = -\log \frac{e^{s^+}}{e^{s^+} + \sum_{k=1}^{K} e^{s_k^-}}
      $
      \begin{itemize}
        \item $s^+ = \mathbf{u}^\top \mathbf{n}^+$: điểm tin dương (đã click)
        \item $s_k^- = \mathbf{u}^\top \mathbf{n}_k^-$: điểm tin âm (không click)
        \item $K=4$: số negative samples mỗi impression
      \end{itemize}

      \vspace{0.3em}
      \begin{keybox}[title={Tương đương cross-entropy $K$-class}]
        \footnotesize
        Bài toán trở thành: ``trong $K{+}1$ tin,
        phân loại đúng tin dương'' $\to$
        tối ưu \textit{ranking} tương đối thay vì
        giá trị tuyệt đối.
      \end{keybox}

    \column{0.50\textwidth}
      \begin{table}
        \footnotesize
        \begin{tabular}{>{\bfseries}lp{4.0cm}c}
          \toprule
          Loss & Đặc điểm & Phù hợp?\\
          \midrule
          BCE & Xét từng (user,news) độc lập, không so sánh & \bad\\[0.3em]
          BPR & So sánh 1 positive vs 1 negative & \textcolor{orange}{\faMinusCircle}\\[0.3em]
          \textbf{InfoNCE} & 1 pos vs $K$ neg \textbf{cùng lúc} & \ok\\
          \bottomrule
        \end{tabular}
      \end{table}
      \vspace{0.3em}
      {\footnotesize
      \textit{BCE} = Binary Cross-Entropy\\
      \textit{BPR} = Bayesian Personalized Ranking\\
      InfoNCE là chuẩn mực của SimCLR, CLIP,
      và NRMS gốc (Wu et al. 2019).}
  \end{columns}
\end{frame}

% ─────────────────────────────────────────────────────────────────────────────
\begin{frame}{Training Pipeline}
  \begin{columns}[T]
    \column{0.50\textwidth}
      \textbf{Optimizer \& Learning Rate:}
      \begin{itemize}
        \item \textbf{AdamW}: Adam + weight decay tách biệt
          $(\text{lr}=10^{-4},\, \lambda=10^{-6})$
        \item Weight decay: regularization tránh overfit,
          không áp dụng cho bias/LayerNorm
      \end{itemize}

      \vspace{0.3em}
      \textbf{Mixed Precision (FP16/FP32):}
      \begin{itemize}
        \item Forward + backward: FP16 (float16)
          $\to$ nhanh gấp $\sim$2x, ít VRAM
        \item Loss scaling: tránh underflow gradient
        \item Optimizer step: FP32 để ổn định
      \end{itemize}

    \column{0.47\textwidth}
      \textbf{Ổn định huấn luyện:}
      \begin{itemize}
        \item \textbf{Gradient Clipping:}
          $\|\nabla\|_{\max} = 1.0$\\
          Cắt gradient nếu norm vượt 1.0,
          tránh gradient explosion khi FP16
        \item \textbf{Dropout} $= 0.2$: tắt ngẫu nhiên
          $20\%$ neuron mỗi forward pass
        \item \textbf{NaN Guard}: \texttt{nan\_to\_num} sau MH-SA
          cho các sequence được pad (đệm) bằng 0
      \end{itemize}

      \vspace{0.2em}
      \textbf{Batching:}
      \begin{itemize}
        \item History pad $L=50$, title pad $T=30$
        \item Negative sampling $K=4$ per impression
      \end{itemize}
  \end{columns}
\end{frame}

% ─────────────────────────────────────────────────────────────────────────────
\begin{frame}{Metrics Đánh Giá --- Ý nghĩa từng chỉ số}
  \begin{columns}[T]
    \column{0.50\textwidth}
      \textbf{AUC} (Area Under ROC Curve):
      \begin{itemize}
        \item Xác suất model xếp tin dương \textit{cao hơn} tin âm
        \item $0.5$ = ngẫu nhiên, $1.0$ = hoàn hảo
        \item Không phụ thuộc ngưỡng quyết định
      \end{itemize}

      \vspace{0.3em}
      \textbf{MRR} (Mean Reciprocal Rank):
      \[
        \text{MRR} = \frac{1}{|\mathcal{U}|}\sum_u \frac{1}{\text{rank}^+_u}
      \]
      Đo vị trí trung bình của tin dương đầu tiên.

    \column{0.47\textwidth}
      \textbf{nDCG@K} (Normalized Discounted Cumulative Gain):
      \[
        \text{nDCG@K} = \frac{\text{DCG@K}}{\text{IDCG@K}}
      \]
      \begin{itemize}
        \item DCG: tổng điểm relevance, chiết khấu $\log_2(\text{rank}+1)$
        \item IDCG: điểm lý tưởng (ranking hoàn hảo)
        \item Phạt nặng khi tin tốt bị đẩy xuống thấp
      \end{itemize}

      \vspace{0.3em}
      \begin{keybox}[]
        \footnotesize
        Dự án báo cáo: AUC, MRR,
        nDCG@5, nDCG@10\\
        --- 4 metrics chuẩn của leaderboard MIND.
      \end{keybox}
  \end{columns}
\end{frame}
